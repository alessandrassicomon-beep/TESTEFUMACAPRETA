    function definirDataHoraPadrao() {
      const hoje = new Date();
      const ano = hoje.getFullYear();
      const mes = String(hoje.getMonth() + 1).padStart(2, '0');
      const dia = String(hoje.getDate()).padStart(2, '0');
      const horas = String(hoje.getHours()).padStart(2, '0');
      const minutos = String(hoje.getMinutes()).padStart(2, '0');

      document.getElementById('ensaioData').value = `${ano}-${mes}-${dia}`;
      document.getElementById('ensaioHora').value = `${horas}:${minutos}`;
    }

    function popularDatalistPlacas() {
      const datalist = document.getElementById('placasDatalist');
      datalist.innerHTML = '';
      VEICULOS_DATA.forEach(v => {
        const opt = document.createElement('option');
        opt.value = v.placa;
        opt.textContent = `${v.placa} - ${v.modelo} (${v.departamento})`;
        datalist.appendChild(opt);
      });
    }

    function calcularResultado() {
      const tipo = document.getElementById('tipoEquipamento').value;
      const densidade = document.getElementById('ensaioDensidade').value;
      const resultBox = document.getElementById('liveResultBox');
      const banner = document.getElementById('liveResultBanner');
      const icon = document.getElementById('liveResultIcon');
      const text = document.getElementById('liveResultText');

      if (!tipo || !densidade) {
        resultBox.style.display = 'none';
        resultadoFinal = "";
        return;
      }

      let conforme = false;
      if (tipo === 'Veículo à Diesel') {
        conforme = (densidade === 'Padrão 1' || densidade === 'Padrão 2');
      } else if (tipo === 'Grupo Motor Gerador') {
        conforme = (densidade === 'Padrão 1');
      }

      resultBox.style.display = 'block';

      if (conforme) {
        resultadoFinal = "CONFORME";
        banner.className = "result-banner result-conforme";
        icon.className = "bi bi-check-circle-fill text-success fs-4";
        text.textContent = "RESULTADO: CONFORME (DENTRO DO LIMITE PERMITIDO)";
      } else {
        resultadoFinal = "NÃO CONFORME";
        banner.className = "result-banner result-nao-conforme";
        icon.className = "bi bi-exclamation-triangle-fill text-danger fs-4";
        text.textContent = "RESULTADO: NÃO CONFORME (EXCEDE O LIMITE PERMITIDO)";
      }
    }

    function getActiveStepElementId(stepNum) {
      if (stepNum === 1) return 'step-1';
      if (stepNum === 2) {
        const tipo = document.getElementById('tipoEquipamento').value;
        return (tipo === 'Veículo à Diesel') ? 'step-2A' : 'step-2B';
      }
      if (stepNum === 3) return 'step-3';
      if (stepNum === 4) {
        return (resultadoFinal === 'CONFORME') ? 'step-4A' : 'step-4B';
      }
      if (stepNum === 5) return 'step-5';
      if (stepNum === 6) return 'step-6';
      if (stepNum === 7) return 'step-7';
      return 'step-1';
    }

    function updateWizardUI() {
      const allSteps = document.querySelectorAll('.form-step');
      allSteps.forEach(el => el.classList.remove('active'));

      const activeId = getActiveStepElementId(currentStep);
      const targetEl = document.getElementById(activeId);
      if (targetEl) targetEl.classList.add('active');

      const stepItems = document.querySelectorAll('.step-item');
      stepItems.forEach(item => {
        const s = parseInt(item.getAttribute('data-step'), 10);
        item.classList.remove('active', 'completed');
        if (s < currentStep) item.classList.add('completed');
        else if (s === currentStep) item.classList.add('active');
      });

      const btnVoltar = document.getElementById('btnVoltar');
      const btnAvancar = document.getElementById('btnAvancar');
      const btnImprimir = document.getElementById('btnImprimir');
      const btnGerarPDF = document.getElementById('btnGerarPDF');

      btnVoltar.style.display = (currentStep === 1) ? 'none' : 'inline-block';

      if (currentStep === 7) {
        btnAvancar.style.display = 'none';
        btnImprimir.style.display = 'inline-block';
        btnGerarPDF.style.display = 'inline-block';
      } else {
        btnAvancar.style.display = 'inline-block';
        btnImprimir.style.display = 'none';
        btnGerarPDF.style.display = 'none';
      }

      window.scrollTo({ top: 0, behavior: 'smooth' });
    }

    function validateCurrentStep() {
      let isValid = true;

      if (currentStep === 1) {
        const tipo = document.getElementById('tipoEquipamento').value;
        const err = document.getElementById('tipoEquipamentoError');
        if (!tipo) {
          err.style.setProperty('display', 'block', 'important');
          isValid = false;
        } else {
          err.style.setProperty('display', 'none', 'important');
        }
      }

      else if (currentStep === 2) {
        const tipo = document.getElementById('tipoEquipamento').value;
        if (tipo === 'Veículo à Diesel') {
          const input = document.getElementById('placaSearchInput');
          if (!selectedVehicle) {
            input.classList.add('is-invalid');
            isValid = false;
          } else {
            input.classList.remove('is-invalid');
          }
        } else {
          const requiredIds = ['geradorLocal', 'geradorSetor', 'geradorFabricante', 'geradorModelo', 'geradorSerie', 'geradorPotencia', 'geradorDataFab'];
          requiredIds.forEach(id => {
            const el = document.getElementById(id);
            if (!el.value.trim()) {
              el.classList.add('is-invalid');
              isValid = false;
            } else {
              el.classList.remove('is-invalid');
            }
          });
        }
      }

      else if (currentStep === 3) {
        const fields = ['ensaioData', 'ensaioHora', 'ensaioDistancia', 'ensaioOperacaoNormal', 'ensaioDensidade'];
        fields.forEach(id => {
          const el = document.getElementById(id);
          if (!el.value.trim()) {
            el.classList.add('is-invalid');
            isValid = false;
          } else {
            el.classList.remove('is-invalid');
          }
        });
        if (isValid) calcularResultado();
      }

      else if (currentStep === 4) {
        if (resultadoFinal === 'NÃO CONFORME') {
          const ncFields = ['ncDescricao', 'ncAcaoCorretiva', 'ncResponsavel', 'ncPrazo'];
          ncFields.forEach(id => {
            const el = document.getElementById(id);
            if (!el.value.trim()) {
              el.classList.add('is-invalid');
              isValid = false;
            } else {
              el.classList.remove('is-invalid');
            }
          });
        }
      }

      else if (currentStep === 7) {
        const respFields = ['respNome', 'respRE'];
        respFields.forEach(id => {
          const el = document.getElementById(id);
          if (!el.value.trim()) {
            el.classList.add('is-invalid');
            isValid = false;
          } else {
            el.classList.remove('is-invalid');
          }
        });

        const sigErr = document.getElementById('sigError');
        if (isCanvasBlank()) {
          sigErr.style.setProperty('display', 'block', 'important');
          isValid = false;
        } else {
          sigErr.style.setProperty('display', 'none', 'important');
        }
      }

      return isValid;
    }

    function nextStep() {
      if (!validateCurrentStep()) return;
      if (currentStep < 7) {
        currentStep++;
        updateWizardUI();
      }
    }

    function prevStep() {
      if (currentStep > 1) {
        currentStep--;
        updateWizardUI();
      }
    }

    function handlePhotoUpload(input, imgId, boxId, iconId, textId, btnRemoverId) {

    }

    /* ASSINATURA CANVAS */
    let canvas, ctx;
    let isDrawing = false;
    let hasDrawn = false;

    function initSignatureCanvas() {
      canvas = document.getElementById('signaturePad');
      ctx = canvas.getContext('2d');

      function resize() {
        const rect = canvas.getBoundingClientRect();
        canvas.width = rect.width;
        canvas.height = rect.height;
        ctx.strokeStyle = "#001B6E";
        ctx.lineWidth = 2.5;
        ctx.lineCap = "round";
        ctx.lineJoin = "round";
      }
      resize();
      window.addEventListener('resize', resize);

      canvas.addEventListener('mousedown', startDrawing);
      canvas.addEventListener('mousemove', draw);
      canvas.addEventListener('mouseup', stopDrawing);
      canvas.addEventListener('mouseleave', stopDrawing);

      canvas.addEventListener('touchstart', (e) => {
        e.preventDefault();
        const touch = e.touches[0];
        const rect = canvas.getBoundingClientRect();
        ctx.beginPath();
        ctx.moveTo(touch.clientX - rect.left, touch.clientY - rect.top);
        isDrawing = true;
        hasDrawn = true;
      });

      canvas.addEventListener('touchmove', (e) => {
        e.preventDefault();
        if (!isDrawing) return;
        const touch = e.touches[0];
        const rect = canvas.getBoundingClientRect();
        ctx.lineTo(touch.clientX - rect.left, touch.clientY - rect.top);
        ctx.stroke();
      });

      canvas.addEventListener('touchend', (e) => {
        e.preventDefault();
        isDrawing = false;
      });
    }

    function startDrawing(e) {
      isDrawing = true;
      hasDrawn = true;
      const rect = canvas.getBoundingClientRect();
      ctx.beginPath();
      ctx.moveTo(e.clientX - rect.left, e.clientY - rect.top);
    }

    function draw(e) {
      if (!isDrawing) return;
      const rect = canvas.getBoundingClientRect();
      ctx.lineTo(e.clientX - rect.left, e.clientY - rect.top);
      ctx.stroke();
    }

    function stopDrawing() { isDrawing = false; }

    function clearSignature() {
      if (ctx && canvas) {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        hasDrawn = false;
      }
    }

    function isCanvasBlank() { return !hasDrawn; }

    function formatarData(isoDate) {
      if (!isoDate) return '';
      const [ano, mes, dia] = isoDate.split('-');
      return `${dia}/${mes}/${ano}`;
    }

    async function gerarPDF() {
      if (!validateCurrentStep()) return;

      const overlay = document.getElementById('loadingOverlay');
      overlay.style.display = 'flex';

      syncPdfData();

      const element = document.getElementById('pdfRenderArea');
      element.style.position = 'static';
      element.style.left = '0';

      try {
        const canvasImage = await html2canvas(element, {
          scale: 2,
          useCORS: true,
          logging: false,
          backgroundColor: '#ffffff'
        });

        element.style.position = 'absolute';
        element.style.left = '-9999px';

        const { jsPDF } = window.jspdf;
        const pdf = new jsPDF('p', 'mm', 'a4');
        const pdfWidth = pdf.internal.pageSize.getWidth();
        const pdfHeight = pdf.internal.pageSize.getHeight();

        const imgData = canvasImage.toDataURL('image/jpeg', 0.95);
        const imgProps = pdf.getImageProperties(imgData);
        const imgHeight = (imgProps.height * pdfWidth) / imgProps.width;

        pdf.addImage(imgData, 'JPEG', 0, 0, pdfWidth, Math.min(imgHeight, pdfHeight));

        const idDoc = selectedVehicle ? selectedVehicle.placa : 'GERADOR';
        pdf.save(`F-SGI-50_Fumaca_Preta_${idDoc}_${document.getElementById('ensaioData').value}.pdf`);

      } catch (error) {
        console.error("Erro ao gerar PDF:", error);
        alert("Ocorreu um erro ao gerar o arquivo PDF. Tente a opção Imprimir.");
      } finally {
        element.style.position = 'absolute';
        element.style.left = '-9999px';
        overlay.style.display = 'none';
      }
    }
